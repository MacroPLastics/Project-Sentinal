#a file to hold all of the debugging stuff for sns

###variables###
network_info=$(ifconfig)

###functions###

#function to call ifconfig
ip_data(){
    $network_info
}

#function to return to main menu

return_to_main_menu() {
    echo "Returning to main menu..."
}

#function to call the editing menus
edit(){
    echo "Starting editors..."
        editor_submenu
            file_submenu
}


###submenus###

editor_submenu(){
    echo "Editor menu."
        echo "1.    kate"
        echo "2.    vim"
        echo "3.    nano"
            read -rp "please choose which editor you would like to use:" editor_choice
                case "$editor_choice" in
                    1)
                        editor="kate -n"
                    ;;
                    2)
                        editor="vim"
                    ;;
                    3)
                        editor="nano"
                    ;;
                    *)
                        echo "Invalid option."
                        editor_submenu
                    ;;
                esac
}

#note when adding this to sns.sh make sure to call the file_submenu after calling the editor!!!!!!!

file_submenu(){
    echo "File menu."
        echo "1.    SentinalNetScanner.sh"
        echo "2.    Scans"
        echo "3     Wireshark"
        echo "4.    Maintenance"
        echo "5.    Debugging"
        echo "q.    Quit"
            read -rp "select file:" file_choice
                case "$file_choice" in
                    [1])
                        echo "Editing SentinalNetScanner.sh with $editor..."
                            $editor "sns.sh"
                    ;;
                    [2])
                        echo "Editing Scans with $editor..."
                            $editor "snsscans.txt"
                    ;;
                    [3])
                        echo "Editing Wireshark with $editor..."
                            $editor "snswireshark.txt"
                    ;;
                    [4])
                        echo "Editing Maintenance with $editor..."
                            $editor "snsmaintenance.txt"
                    ;;
                    [5])
                        echo "Editing Debugging with $editor..."
                            $editor "snsdebugging.txt"
                    ;;
                    [Qq])
                        echo "Exiting..."
                            exit 0
                    ;;
                    *)
                        echo "Invalid option."
                            file_submenu
                    ;;
                esac
}
